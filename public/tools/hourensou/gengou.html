④ プログラムの {例|れい}

JavaScript で {書|か}いた {例|れい}です。{考|かんが}え{方|かた}は 3つだけです。

  1. ① {年|ねん}・{月|がつ}・{日|にち}を 1つの {数|かず}に する
    2019{年|ねん}5{月|がつ}1{日|にち} → 2019 × 10000 ＋ 5 × 100 ＋ 1 → 20190501
  2. ② {新|あたら}しい {元号|げんごう}から じゅんに、はじまりの {日|ひ}と くらべる

  3. ③ {何年目|なんねんめ}かを {計算|けいさん}する
    2019 − 2019 ＋ 1 ＝ 1 → 1{年目|ねんめ}は「{元年|がんねん}」 
// Era table (newest first): name, first day as YYYYMMDD, first year
const ERAS = [
  { name: "令和", start: 20190501, firstYear: 2019 },
  { name: "平成", start: 19890108, firstYear: 1989 },
  { name: "昭和", start: 19261225, firstYear: 1926 },
];

// Western date -> Japanese era date (2019, 5, 1 -> "令和元年5月1日")
function toWareki(year, month, day) {
  const ymd = year * 10000 + month * 100 + day; // step 1: 20190501
  for (const era of ERAS) {
    if (ymd >= era.start) {                     // step 2: compare
      const n = year - era.firstYear + 1;       // step 3: n-th year
      const nen = n === 1 ? "元" : String(n);   // 1st year = "元年"
      return era.name + nen + "年" + month + "月" + day + "日";
    }
  }
  return "out of range (before Showa)";
}

console.log(toWareki(2019, 4, 30)); // 平成31年4月30日
console.log(toWareki(2019, 5, 1));  // 令和元年5月1日
console.log(toWareki(1989, 1, 7));  // 昭和64年1月7日
